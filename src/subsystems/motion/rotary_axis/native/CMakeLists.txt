cmake_minimum_required(VERSION 3.16)
project(simulatorx_reference_sdk C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The reference SDK bridge requires Linux Unix domain sockets")
endif()
add_library(simulatorx_sdk SHARED simulatorx_sdk.c)
set_target_properties(simulatorx_sdk PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_VISIBILITY_PRESET hidden)
target_compile_options(simulatorx_sdk PRIVATE -Wall -Wextra -Werror)
target_include_directories(simulatorx_sdk PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
